from database import Base #Se importa la base que es una clase que extiende de base de datos relacional
from sqlalchemy import Column, Integer, String, ForeignKey
#Se integran los datos para poder usarlos como 
#tipos de datos en el codigo
from sqlalchemy.orm import relationship

#Se crea la tabla jugador como una clase, con su nombre
class Jugador(Base):
  __tablename__ = "jugadores"
#atributos de la case jugador, los cuales con el ID que se ponen con los atributos necesarios
  id = Column(Integer, primary_key=True, index=True, autoincrement=True)
  nombre = Column(String(100), nullable=False)
  partidas_ganadas = Column(Integer, default = 0, nullable = False)
  
  #Relaciones
  movimientos = relationship("LogMovimiento", back_populates="jugador")
  partidas_ganadas_rel = relationship("Partida", foreign_keys="Partida.ganador_id", back_populates="ganador")
  participaciones = relationship("JugadorPartida", back_populates="jugador")

class LogMovimiento(Base):
  __tablename__ = "log_movimientos"

  id = Column(Integer, primary_key=True, index=True, autoincrement=True)
  partida_id = Column(Integer, ForeignKey("partidas.id"))
  jugador_id = Column(Integer, ForeignKey("jugadores.id"))
  accion = Column(String(255), nullable=False)
  
  jugador = relationship("Jugador", back_populates="movimientos")
  partida = relationship("Partida", back_populates="movimientos")
  
class Partida(Base):
    __tablename__ = "partidas"
    
    id = Column(Integer, primary_key=True, index=True, autoincrement=True)
    ganador_id = Column(Integer, ForeignKey("jugadores.id"), nullable=True)
    
    ganador = relationship("Jugador", foreign_keys=[ganador_id], back_populates="partidas_ganadas_rel")
    movimientos = relationship("LogMovimiento", back_populates="partida")
    jugadores = relationship("JugadorPartida", back_populates="partida")

class JugadorPartida(Base):
    __tablename__ = "jugador_partida"
    id = Column(Integer, primary_key=True, index=True, autoincrement=True)
    jugador_id = Column(Integer, ForeignKey("jugadores.id"), nullable=False)
    partida_id = Column(Integer, ForeignKey("partidas.id"), nullable=False)
    
    jugador = relationship("Jugador", back_populates="participaciones")
    partida = relationship("Partida", back_populates="jugadores")

using System;
using Uno.ViewModels;

namespace Uno.Models;

public enum CardColor { Red, Yellow, Green, Blue, Wild }
public enum CardValue { Zero, One, Two, Three, Four, Five, Six, Seven, Eight, Nine, Skip, Reverse, DrawTwo, Wild, WildDrawFour }

public class Card : ViewModelBase 
{
    public CardColor Color { get; set; }
    public CardValue Value { get; set; }

    // Propiedad que indica si la carta es válida para jugarse ahorita
    private bool _esJugable;
    public bool esJugable
    {
        get => _esJugable;
        set => SetProperty(ref _esJugable, value);
    }

    public string ImagePath => GetImagePath();
    public Action<Card>? PlayAction { get; set; }

    public Card(CardColor color, CardValue value)
    {
        Color = color;
        Value = value;
    }

    public bool Jugable(Card topCard, CardColor currentActiveColor)
    {
        if (Color == CardColor.Wild) return true;
        if (Color == currentActiveColor) return true;
        if (Value == topCard.Value) return true;
        return false;
    }

    public void OnCardClicked()
    {
        PlayAction?.Invoke(this);
    }

    private string GetImagePath()
    {
        if (Color == CardColor.Wild)
        {
            return Value == CardValue.Wild
                ? "avares://Uno/Assets/Cards/wild.png"
                : "avares://Uno/Assets/Cards/wild_drawfour.png";
        }

        string colorStr = Color.ToString().ToLower();
        string valStr = Value switch
        {
            CardValue.Zero => "0",
            CardValue.One => "1",
            CardValue.Two => "2",
            CardValue.Three => "3",
            CardValue.Four => "4",
            CardValue.Five => "5",
            CardValue.Six => "6",
            CardValue.Seven => "7",
            CardValue.Eight => "8",
            CardValue.Nine => "9",
            CardValue.Skip => "_skip",
            CardValue.Reverse => "_reverse",
            CardValue.DrawTwo => "_drawtwo",
            _ => ""
        };

        return $"avares://Uno/Assets/Cards/{colorStr}{valStr}.png";
    }

    public override string ToString() => $"{Color} {Value}";
}

using System;
using System.Collections.Generic;
using System.Linq;
using System.Text.Json;
using Uno.Services;
using Uno.Models;
using System.Threading.Tasks;

namespace Uno.Models;

public class UnoGame
{
    public List<Player> Jugadores { get; private set; } = new();
    public List<Card> Pila { get; private set; } = new();
    public List<Card> PilaJugada { get; private set; } = new();

    public int actualJugadorIndex { get; private set; } = 0;
    public int Direccion { get; private set; } = 1;
    public CardColor ColorActual { get; set; }

    public int PilaPendiente { get; set; } = 0; 
    public int PasesContinuos { get; set; } = 0;
    public int partidaID = 1;

    public Card TopCard => PilaJugada.LastOrDefault()!;
    public Player actualJugador => Jugadores[actualJugadorIndex];

    public UnoGame()
    {
        InitializeDeck();
        StartGame();
    }
    public UnoGame(List<Player> players, int id)
    {
        partidaID = id;
        Jugadores = players;
        InitializeDeck();
        StartGame();
    }

    private void InitializeDeck()
    {
        Pila.Clear();
        PilaJugada.Clear();
        
        // 108 Cartas Exactas de UNO
        foreach (CardColor color in Enum.GetValues(typeof(CardColor)))
        {
            if (color == CardColor.Wild) continue;

            Pila.Add(new Card(color, CardValue.Zero)); // 1 cero por color (4)

            for (int i = 0; i < 2; i++) // 2 de cada una (1-9, Skip, Reverse, DrawTwo) (18 * 4 = 72)
            {
                foreach (CardValue val in Enum.GetValues(typeof(CardValue)))
                {
                    if (val == CardValue.Zero || val == CardValue.Wild || val == CardValue.WildDrawFour) continue;
                    Pila.Add(new Card(color, val));
                }
            }
        }

        for (int i = 0; i < 4; i++) // 4 Comodines normales y 4 Comodines +4 (8)
        {
            Pila.Add(new Card(CardColor.Wild, CardValue.Wild));
            Pila.Add(new Card(CardColor.Wild, CardValue.WildDrawFour));
        }

        Shuffle();
    }

    private void Shuffle()
    {
        Random rnd = new();
        Pila = Pila.OrderBy(_ => rnd.Next()).ToList();
    }

    public void StartGame()
    {
        foreach (var player in Jugadores)
        {
            for (int i = 0; i < 7; i++)
            {
                if (Pila.Count > 0)
                {
                    
                    entregar_carta(player);
                }
            }
        }

        Card primeraCarta;
        do
        {
            primeraCarta = TomarCartaDePila()!;
            PilaJugada.Add(primeraCarta);
        } while (primeraCarta.Color == CardColor.Wild);

        ColorActual = primeraCarta.Color;
    }
    
    public async void entregar_carta(Player player)
    {
        
        Card? card = TomarCartaDePila();
        if (card == null)  return;
        
        player.Mano.Add(TomarCartaDePila()!);
        await RegistrarRoboEnServidor(player, card);
    }

    public Card? TomarCartaDePila()
    {
        if (Pila.Count == 0)
        {
            return null; // Mazo agotado (sin rebarajear)
        }

        Card actualCarta = Pila[0];
        Pila.RemoveAt(0);
        if (Pila.Count == 0){
            PilaTermino();
        }

        return actualCarta;
    }

    public bool JugarCarta(Player jugador, Card carta, CardColor? actualColor = null)
    {
        if (jugador != actualJugador) return false;
        if (!jugador.Mano.Contains(carta)) return false;

        // Regla de acumulación de +2 / +4
        if (PilaPendiente > 0)
        {
            if (carta.Value != CardValue.DrawTwo && carta.Value != CardValue.WildDrawFour)
                return false; // Si hay castigo acumulado, solo puede responder con +2 o +4
        }
        else if (!carta.Jugable(TopCard, ColorActual))
        {
            return false;
        }

        jugador.Mano.Remove(carta);
        PilaJugada.Add(carta);
        PasesContinuos = 0; // Se resetea el contador de pases

        ColorActual = carta.Color == CardColor.Wild ? (actualColor ?? CardColor.Red) : carta.Color;

        switch (carta.Value)
        {
            case CardValue.DrawTwo:
                PilaPendiente += 2;
                break;
            case CardValue.WildDrawFour:
                PilaPendiente += 4;
                break;
            case CardValue.Skip:
                SiguienteTurno();
                break;
            case CardValue.Reverse:
                Direccion *= -1;
                break;
        }

        SiguienteTurno();
        return true;
    }

    public void ResuelveTomarPendientes(Player player)
    {
        for (int i = 0; i < PilaPendiente; i++)
        {
            Card? actualCarta = TomarCartaDePila();
            if (actualCarta != null) player.Mano.Add(actualCarta);
        }
        PilaPendiente = 0;
        SiguienteTurno();
    }

    public int CalcularPuntos(Player player)
    {
        int puntos = 0;
        foreach (var carta in player.Mano)
        {
            puntos += carta.Value switch
            {
                CardValue.Zero => 0,
                CardValue.One => 1,
                CardValue.Two => 2,
                CardValue.Three => 3,
                CardValue.Four => 4,
                CardValue.Five => 5,
                CardValue.Six => 6,
                CardValue.Seven => 7,
                CardValue.Eight => 8,
                CardValue.Nine => 9,
                _ => 0 // Comodines y especiales valen 0
            };
        }
        return puntos;
    }

    public void SiguienteTurno()
    {
        actualJugadorIndex = (actualJugadorIndex + Direccion + Jugadores.Count) % Jugadores.Count;
    }

namespace Uno.Services;
using System;
using System.Net.Http;
using System.Text;
using System.Text.Json;
using System.Threading.Tasks;
using Uno.Models;

//El servicio de la API es basicamente para poder centralizar las llamadas que se le hacen y para no tener que hacer
//todo el protocolo de llamadas cada que se quiere hacer una peticion por ejemplo de post, solo se le llama a esta 
//funcion, se manda en que parte de la url se va a ocupar o que terminal va a tener y los datos que se le pasan o 
//los datos que va a recibir

public class ApiService
{
    //Se declara la base de la url con la que se va a conectar con la base de datos
    public static string ServerIP { get; set; } = "127.0.0.1";
    public static string BaseURL => $"http://{ServerIP}:8000";

    //Un solo HttpClient compartido por toda la aplicacion (crear uno nuevo en cada peticion desperdicia sockets).
    //Si el servidor no contesta en 5 segundos la peticion se cancela y el juego sigue sin registrar ese movimiento
    private static readonly HttpClient _client = new HttpClient { Timeout = TimeSpan.FromSeconds(5) };

    //Funcion que envia las jugadas o movimientos, tambien podria enviar en un futuro a los jugadores
    public async Task<string?> EnviarJugadaLog(string endpoint, object data)
    {
        try
        {
            //hace el objeto que le mandamos en formato JSON
            string jsonMessage = JsonSerializer.Serialize(data);

            //ahora que ya esta en formato JSON se hace conte
            HttpContent content = new StringContent(jsonMessage, Encoding.UTF8, "application/json");

            string url = BaseURL + endpoint;
            HttpResponseMessage response = await _client.PostAsync(url, content);
            response.EnsureSuccessStatusCode();

            string respuestaServidor = await response.Content.ReadAsStringAsync();
            return respuestaServidor;
        }
        catch (Exception e)
        {
            //Se atrapa cualquier excepcion (servidor apagado, tiempo agotado, etc.) porque estas funciones se llaman
            //desde metodos async void y una excepcion sin atrapar cerraria toda la aplicacion
            Console.WriteLine(e.Message);
            return null;
        }
    }

    public async Task<string?> Obtenerdatos(string endpoint)
    {
        try
        {
            string url = BaseURL + endpoint;

            HttpResponseMessage response = await _client.GetAsync(url);
            response.EnsureSuccessStatusCode();

            return await response.Content.ReadAsStringAsync();
        }
        catch (Exception e)
        {
            Console.WriteLine(e.Message);
            return null;
        }
    }
}

using System;
using System.Collections.Generic;
using System.Globalization;
using Avalonia.Data.Converters;
using Avalonia.Media.Imaging;
using Avalonia.Platform;
namespace Uno.ViewModels;

public class BitmapAssetValueConverter : IValueConverter
{
    public static readonly BitmapAssetValueConverter Instance = new();
    private static readonly Dictionary<string, Bitmap> _cache = new();
    public object? Convert(object? value, Type targetType, object? parameter, CultureInfo culture)
    {
        if (value is string rawUri && !string.IsNullOrEmpty(rawUri))
        {
            try
            {
                if (_cache.TryGetValue(rawUri, out var cached)) return cached;
                var uri = new Uri(rawUri);
                if (AssetLoader.Exists(uri))
                {
                    using var stream = AssetLoader.Open(uri);
                    var bitmap = new Bitmap(stream);
                    _cache[rawUri] = bitmap;
                    return bitmap;
                }
            }
            catch
            {
                return null;
            }
        }
        return null;
    }

    public object ConvertBack(object? value, Type targetType, object? parameter, CultureInfo culture)
    {
        throw new NotSupportedException();
    }
}

    public event PropertyChangedEventHandler? PropertyChanged;

    protected void OnPropertyChanged([CallerMemberName] string? propertyName = null)
    {
        PropertyChanged?.Invoke(this, new PropertyChangedEventArgs(propertyName));
    }

    protected bool SetProperty<T>(ref T field, T value, [CallerMemberName] string? propertyName = null)
    {
        if (Equals(field, value)) return false;
        field = value;

        private void EfectoHoverEnMano(object? sender, PointerWheelEventArgs e)
        {
            if (sender is ScrollViewer scrollViewer)
            {
                if (e.Delta.Y != 0)
                {
                    double scrollVel = 40; 
                    double newXOffset = scrollViewer.Offset.X - (e.Delta.Y * scrollVel);
                    scrollViewer.Offset = new Avalonia.Vector(newXOffset, scrollViewer.Offset.Y);
                    e.Handled = true;
                }
            }

using System;
using System.Diagnostics.CodeAnalysis;
using Avalonia.Controls;
using Avalonia.Controls.Templates;
using Uno.ViewModels;

namespace Uno;

/// <summary>
/// Given a view model, returns the corresponding view if possible.
/// </summary>
[RequiresUnreferencedCode(
    "Default implementation of ViewLocator involves reflection which may be trimmed away.",
    Url = "https://docs.avaloniaui.net/docs/concepts/view-locator")]
public class ViewLocator : IDataTemplate
{
    public Control? Build(object? param)
    {
        if (param is null)
            return null;

        var name = param.GetType().FullName!.Replace("ViewModel", "View", StringComparison.Ordinal);
        var type = Type.GetType(name);

        if (type != null)
        {
            return (Control)Activator.CreateInstance(type)!;
        }

        return new TextBlock { Text = "Not Found: " + name };
    }

    public bool Match(object? data)
    {
        return data is ViewModelBase;
    }
}
